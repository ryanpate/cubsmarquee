# FlightWall-Style Flight Display Restyle Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Restyle all flight-tracking screens in `flight_display.py` to the FlightWall Mini aesthetic: black background, 20x20 pixel-art airline logo, mixed-case text, white labels with cyan values.

**Architecture:** Pure visual restyle — no data-fetching, rotation, or timing logic changes. New assets (`logos/airlines/*.png`) are generated by a committed tool script. A headless render harness captures one PNG frame per screen via `ScoreboardManager`'s existing PIL preview mirror (rgbmatrix mocked like `tests/conftest.py`), so every task ends with reviewable images.

**Tech Stack:** Python 3.9+ (`from __future__ import annotations`), PIL/Pillow, BDF bitmap fonts, pytest with mocked `rgbmatrix`.

**Spec:** `docs/superpowers/specs/2026-07-31-flightwall-restyle-design.md`

## Global Constraints

- Matrix is 96x48 (`DisplayConfig.MATRIX_COLS` x `DisplayConfig.MATRIX_ROWS`). `draw_text(font_name, x, y, color, text)` — **y is the text baseline**, not the top.
- Fonts (already registered in `ScoreboardManager.FONT_MAPPING`): `'small'` = 6x9, `'tiny'` = 5x7, `'micro'` = 4x6. Char widths: small 6, tiny 5, micro 4.
- New colors: `Colors.FLIGHT_CYAN = (120, 220, 255)`, `Colors.FLIGHT_DIM = (90, 90, 90)`.
- Mixed-case rule: display strings are `.title()`-cased only when longer than 3 chars (so `UPS`, `ANA`, IATA codes stay uppercase).
- Radar constraint: every non-aircraft `draw_pixel` on the radar scope must keep its **red component < 127** — `tests/test_features.py::TestRadarOneDotPerPlane` attributes pixels with red >= 127 to aircraft.
- No changes to: data fetching, route enrichment, caches, rotation/timing logic, `get_quick_flight_summary`, constructor signature `FlightDisplay(scoreboard_manager)` (used by `off_season_handler.py:49` and `live_game_handler.py:374`).
- Run tests from the repo root (`/Users/ryanpate/cubsmarquee`) — code uses relative paths like `./fonts`, `./logos`.

  `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>` and
  `Claude-Session: https://claude.ai/code/session_01LW2JHK92nY5VcAmmYoAkst`

"""Render one PNG frame of each flight screen for visual review.

Runs headless: rgbmatrix is mocked exactly like tests/conftest.py, and

class _FrameCaptured(Exception):
    """Raised from a patched swap_canvas to stop after one frame"""


def _make_display(manager):
    from flight_display import FlightDisplay
    from scoreboard_config import Colors

    d = FlightDisplay.__new__(FlightDisplay)
    d.manager = manager
    d.latitude, d.longitude = 41.9484, -87.6553
    d.flight_max_range_nm = 30
    d.enable_flight_radar = True
    d.flight_data = [dict(f) for f in FIXTURE_FLIGHTS]

    d.FLIGHT_BLUE = Colors.FLIGHT_BLUE
    d.FLIGHT_DARK_BLUE = Colors.FLIGHT_DARK_BLUE

    d.ALTITUDE_HIGH = Colors.FLIGHT_ALTITUDE_HIGH

    d.ALTITUDE_LOW = Colors.FLIGHT_ALTITUDE_LOW
    d.FLIGHT_CYAN = Colors.FLIGHT_CYAN
    # Pre-restyle only: the gradient header cache (removed in Task 6)
    if hasattr(d, '_create_flight_header_background'):
        d._flight_header_bg = d._create_flight_header_background()
    # Post-Task-2 only: airline logos
    if hasattr(d, '_load_airline_logos'):
        d.airline_logos = d._load_airline_logos()
    return d


def _save(manager, name):
    frame = manager.get_frame_copy()
    frame.save(os.path.join(OUT_DIR, f'{name}.png'))
    frame.resize((96 * 8, 48 * 8), Image.NEAREST).save(
        os.path.join(OUT_DIR, f'{name}@8x.png'))
    print(f'  wrote {name}.png')


def _capture(manager, name, fn):
    """Run fn, stopping at the first swap_canvas, and save the frame"""
    def stop(*args, **kwargs):
        raise _FrameCaptured
    manager.swap_canvas = stop
    try:
        fn()
    except _FrameCaptured:
        pass
    finally:
        del manager.swap_canvas  # restore the real method
    _save(manager, name)


def main() -> None:
    os.makedirs(OUT_DIR, exist_ok=True)
    from scoreboard_manager import ScoreboardManager

    manager = ScoreboardManager()
    d = _make_display(manager)

    _capture(manager, 'detail_a',
             lambda: d._draw_detail_frame(d.flight_data[0], '1/3', 0.0))
    _capture(manager, 'detail_b',
             lambda: d._draw_detail_frame(d.flight_data[0], '1/3', 4.0))
    _capture(manager, 'detail_ga',
             lambda: d._draw_detail_frame(d.flight_data[2], '3/3', 4.0))
    _capture(manager, 'summary', lambda: d._display_summary_view(5))
    _capture(manager, 'radar', lambda: d._display_radar_view(0, 5))

    d.flight_data = []
    _capture(manager, 'no_flights', lambda: d._display_no_flights(5))
    print(f'Done. Previews in {OUT_DIR}')


if __name__ == '__main__':
    main()

def _pil_font(bdf_path: str):
    os.makedirs(PIL_FONT_DIR, exist_ok=True)
    base = os.path.join(
        PIL_FONT_DIR, os.path.splitext(os.path.basename(bdf_path))[0])
    if not os.path.exists(base + '.pil'):
        with open(bdf_path, 'rb') as fp:
            BdfFontFile.BdfFontFile(fp).save(base)
    return ImageFont.load(base + '.pil')


def from_pixel_map(palette: dict, rows: list[str]) -> Image.Image:
    img = Image.new('RGB', (20, 20), (0, 0, 0))
    px = img.load()
    assert len(rows) == 20, f'need 20 rows, got {len(rows)}'
    for y, row in enumerate(rows):
        assert len(row) == 20, f'row {y} is {len(row)} chars, need 20'
        for x, ch in enumerate(row):
            if ch != '.':
                px[x, y] = palette[ch]
    return img


def letter_badge(text: str, bg, fgs) -> Image.Image:
    img = Image.new('RGB', (20, 20), (0, 0, 0))
    draw = ImageDraw.Draw(img)
    try:
        draw.rounded_rectangle((0, 0, 19, 19), radius=5, fill=bg)
    except AttributeError:  # Pillow < 8.2
        draw.rectangle((0, 0, 19, 19), fill=bg)
    font = _pil_font('./fonts/5x8.bdf')
    x = (20 - 5 * len(text)) // 2
    for ch, fg in zip(text, fgs):
        draw.text((x, 6), ch, font=font, fill=fg)
        x += 5
    return img


def main() -> None:
    os.makedirs(OUT_DIR, exist_ok=True)
    logos: dict[str, Image.Image] = {}
    for name, (palette, rows) in PIXEL_LOGOS.items():
        logos[name] = from_pixel_map(palette, rows)
    for name, (text, bg, fgs) in LETTER_BADGES.items():
        logos[name] = letter_badge(text, bg, fgs)

    for name, img in sorted(logos.items()):
        img.save(os.path.join(OUT_DIR, f'{name}.png'))
        print(f'  wrote {name}.png')

    # 8x contact sheet for visual review
    os.makedirs(SHEET_DIR, exist_ok=True)
    names = sorted(logos)
    sheet = Image.new('RGB', (len(names) * 24 * 8, 24 * 8), (30, 30, 30))
    for i, name in enumerate(names):
        big = logos[name].resize((160, 160), Image.NEAREST)
        sheet.paste(big, (i * 24 * 8 + 16, 16))
    sheet.save(os.path.join(SHEET_DIR, 'logo_sheet.png'))
    print(f'Done: {len(logos)} logos + logo_sheet.png')


if __name__ == '__main__':
    main()

```python
    @staticmethod
    def _fmt_alt(altitude_ft: int) -> str:
        """732 -> '732ft', 4100 -> '4.1kft', 34000 -> '34kft'"""
        if altitude_ft < 1000:
            return f"{altitude_ft}ft"
        if altitude_ft < 10000:
            return f"{altitude_ft / 1000:.1f}kft"
        return f"{round(altitude_ft / 1000)}kft"

    @staticmethod
    def _display_case(name: str) -> str:
        """Mixed case for display; short codes (UPS, ANA, LAX) stay upper"""
        return name.title() if len(name) > 3 else name

    def _friendly_type(self, type_code: str | None) -> str:
        """Marketing name for an ICAO type code ('A21N' -> 'A321neo')"""
        if not type_code:
            return ''
        return self.TYPE_NAMES.get(type_code.upper(), type_code)

    def _draw_segments(
        self, font_name: str, x: int, y: int,
        segments: list[tuple[str, RGBColor]],
    ) -> int:
        """Draw (text, color) runs left to right; returns the final x"""
        char_w = {'small': 6, 'tiny': 5, 'tiny_bold': 5, 'micro': 4}[font_name]
        for text, color in segments:
            self.manager.draw_text(font_name, x, y, color, text)
            x += len(text) * char_w
        return x

    def _metric_segments(
        self, flight: dict[str, Any]
    ) -> tuple[list[tuple[str, RGBColor]], list[tuple[str, RGBColor]]]:
        """The two FlightWall metric lines as (text, color) segments"""
        white, cyan = self.FLIGHT_WHITE, self.FLIGHT_CYAN
        heading = flight.get('heading')
        vertical_rate = flight.get('vertical_rate')
        trk = f"{round(heading) % 360}deg" if heading is not None else '---'
        vr = f"{int(vertical_rate)}fpm" if vertical_rate is not None else '---'
        line1 = [('Alt:', white), (self._fmt_alt(flight['altitude_ft']), cyan),
                 (' Spd:', white), (f"{flight['velocity_mph']}mph", cyan)]
        line2 = [('Trk:', white), (trk, cyan), (' Vr:', white), (vr, cyan)]
        return line1, line2
```
